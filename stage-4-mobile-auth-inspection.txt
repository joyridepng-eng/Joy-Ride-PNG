import { RideMap } from "@/components/ride-map";
import { API_BASE_URL } from "@/constants/api";
import { useDriverLocationTracking } from "@/hooks/use-driver-location-tracking";
import { useDrivingRoute } from "@/hooks/use-driving-route";
import { useEffect, useMemo, useState } from "react";
import {
  Platform,
  ScrollView,
  StyleSheet,
  Text,
  TextInput,
  TouchableOpacity,
  View,
} from "react-native";

type Ride = {
  id: string;
  status: string;
  riderName: string;
  riderPhone: string;
  pickupLabel: string | null;
  pickupAddress: string | null;
  pickupLatitude: number | null;
  pickupLongitude: number | null;
  dropoffLabel: string | null;
  dropoffAddress: string | null;
  dropoffLatitude: number | null;
  dropoffLongitude: number | null;
  estimatedFare: number | null;
  finalFare: number | null;
  driver?: {
    id: string;
    fullName: string;
    phone: string;
  } | null;
  vehicle?: {
    id: string;
    plateNumber: string;
    make: string;
    model: string;
    color: string;
  } | null;
};

const ACTIVE_DRIVER_STATUSES = ["ACCEPTED", "DRIVER_ARRIVED", "IN_PROGRESS"];

export default function AssignedRideScreen() {
  const driverId = "cmqxiib050000dcrih388jp0a";
  const [assignedRide, setAssignedRide] = useState<Ride | null>(null);
  const [message, setMessage] = useState("");
  const [isLoading, setIsLoading] = useState(false);
  const [isUpdatingStatus, setIsUpdatingStatus] = useState(false);

  const trackingEnabled =
    Platform.OS !== "web" &&
    assignedRide !== null &&
    ACTIVE_DRIVER_STATUSES.includes(assignedRide.status);

  const {
    status: trackingStatus,
    errorMessage: trackingError,
    latestLocation,
    isTracking,
  } = useDriverLocationTracking({
    driverId,
    rideId: assignedRide?.id,
    enabled: trackingEnabled,
  });

  const driverMapLocation = useMemo(
    () =>
      latestLocation
        ? {
            latitude: latestLocation.latitude,
            longitude: latestLocation.longitude,
            accuracy: latestLocation.accuracy,
            heading: latestLocation.heading,
            speed: latestLocation.speed,
            label: "Your live location",
          }
        : null,
    [
      latestLocation?.latitude,
      latestLocation?.longitude,
      latestLocation?.accuracy,
      latestLocation?.heading,
      latestLocation?.speed,
    ],
  );

  const pickupMapLocation = useMemo(
    () =>
      assignedRide?.pickupLatitude !== null &&
      assignedRide?.pickupLatitude !== undefined &&
      assignedRide?.pickupLongitude !== null &&
      assignedRide?.pickupLongitude !== undefined
        ? {
            latitude: assignedRide.pickupLatitude,
            longitude: assignedRide.pickupLongitude,
            label:
              assignedRide.pickupLabel ||
              "Pickup",
            address:
              assignedRide.pickupAddress ||
              assignedRide.pickupLabel ||
              "Pickup location",
          }
        : null,
    [
      assignedRide?.pickupLatitude,
      assignedRide?.pickupLongitude,
      assignedRide?.pickupLabel,
      assignedRide?.pickupAddress,
    ],
  );

  const dropoffMapLocation = useMemo(
    () =>
      assignedRide?.dropoffLatitude !== null &&
      assignedRide?.dropoffLatitude !== undefined &&
      assignedRide?.dropoffLongitude !== null &&
      assignedRide?.dropoffLongitude !== undefined
        ? {
            latitude: assignedRide.dropoffLatitude,
            longitude: assignedRide.dropoffLongitude,
            label:
              assignedRide.dropoffLabel ||
              "Drop-off",
            address:
              assignedRide.dropoffAddress ||
              assignedRide.dropoffLabel ||
              "Drop-off location",
          }
        : null,
    [
      assignedRide?.dropoffLatitude,
      assignedRide?.dropoffLongitude,
      assignedRide?.dropoffLabel,
      assignedRide?.dropoffAddress,
    ],
  );

  const navigationMode =
    assignedRide?.status === "IN_PROGRESS"
      ? "dropoff"
      : assignedRide?.status === "DRIVER_ARRIVED"
        ? "waiting"
        : "pickup";

  const activeDestination =
    navigationMode === "dropoff"
      ? dropoffMapLocation
      : pickupMapLocation;

  const {
    route: drivingRoute,
    errorMessage: routeError,
    isLoading: routeLoading,
  } = useDrivingRoute({
    origin: driverMapLocation,
    destination: activeDestination,
    enabled:
      trackingEnabled &&
      driverMapLocation !== null &&
      activeDestination !== null,
  });

  useEffect(() => {
    if (Platform.OS === "web") {
      return;
    }

    if (!assignedRide) {
      return;
    }

    if (!ACTIVE_DRIVER_STATUSES.includes(assignedRide.status)) {
      return;
    }

    const interval = setInterval(() => {
      findAssignedRide(true);
    }, 5000);

    return () => clearInterval(interval);
  }, [assignedRide?.id, assignedRide?.status]);

  async function findAssignedRide(isAutoRefresh = false) {

    try {
      if (!isAutoRefresh) {
        setIsLoading(true);
        setMessage("");
        setAssignedRide(null);
      }

      const response = await fetch(`${API_BASE_URL}/rides`);
      const data = await response.json();

      if (!response.ok) {
        if (!isAutoRefresh) {
          setMessage(data.message || "Could not load rides.");
        }
        return;
      }

      const rides = Array.isArray(data) ? data : [];

      const matchedRide =
        rides.find(
          (ride: Ride) =>
            ride.driver?.id === driverId.trim() &&
            ACTIVE_DRIVER_STATUSES.includes(ride.status),
        ) || null;

      if (!matchedRide) {
        setAssignedRide(null);

        if (!isAutoRefresh) {
          setMessage("No active assigned ride found for this driver.");
        }

        return;
      }

      setAssignedRide(matchedRide);

      if (!isAutoRefresh) {
        setMessage("Assigned ride loaded.");
      }
    } catch {
      if (!isAutoRefresh) {
        setMessage(
          `Could not connect to the backend. Make sure the backend server is running on ${API_BASE_URL}.`,
        );
      }
    } finally {
      if (!isAutoRefresh) {
        setIsLoading(false);
      }
    }
  }

  async function updateRideStatus(nextStatus: string) {
    if (!assignedRide) {
      setMessage("No assigned ride selected.");
      return;
    }

    try {
      setIsUpdatingStatus(true);
      setMessage("");

      const response = await fetch(
        `${API_BASE_URL}/rides/${assignedRide.id}/status`,
        {
          method: "PATCH",
          headers: {
            "Content-Type": "application/json",
          },
          body: JSON.stringify({
            status: nextStatus,
          }),
        },
      );

      const data = await response.json();

      if (!response.ok) {
        setMessage(data.message || "Could not update ride status.");
        return;
      }

      if (ACTIVE_DRIVER_STATUSES.includes(data.status)) {
        setAssignedRide(data);
        setMessage(`Ride updated to ${data.status}.`);
      } else {
        setAssignedRide(null);
        setMessage(
          `Ride updated to ${data.status}. This ride is no longer active.`,
        );
      }
    } catch {
      setMessage(
        `Could not connect to the backend. Make sure the backend server is running on ${API_BASE_URL}.`,
      );
    } finally {
      setIsUpdatingStatus(false);
    }
  }

  function getDriverActionMessage(status: string) {
    switch (status) {
      case "ACCEPTED":
        return "Go to the pickup location and mark arrived when you reach the rider.";
      case "DRIVER_ARRIVED":
        return "You have arrived. Start the ride when the rider is in the vehicle.";
      case "IN_PROGRESS":
        return "Ride is in progress. Complete the ride when you reach the dropoff location.";
      default:
        return "This ride is not active for driver action.";
    }
  }

  function getAutoRefreshMessage(status: string) {
    if (!ACTIVE_DRIVER_STATUSES.includes(status)) {
      return "Auto-refresh stopped because this ride is no longer active.";
    }

    return "Auto-refresh is active. This screen checks for assigned ride updates every 5 seconds.";
  }

  function renderActionButton() {
    if (!assignedRide) {
      return null;
    }

    if (assignedRide.status === "ACCEPTED") {
      return (
        <TouchableOpacity
          style={[
            styles.actionButton,
            isUpdatingStatus ? styles.disabledButton : null,
          ]}
          onPress={() => updateRideStatus("DRIVER_ARRIVED")}
          disabled={isUpdatingStatus}
        >
          <Text style={styles.actionButtonText}>
            {isUpdatingStatus ? "Updating..." : "Mark Arrived"}
          </Text>
        </TouchableOpacity>
      );
    }

    if (assignedRide.status === "DRIVER_ARRIVED") {
      return (
        <TouchableOpacity
          style={[
            styles.actionButton,
            isUpdatingStatus ? styles.disabledButton : null,
          ]}
          onPress={() => updateRideStatus("IN_PROGRESS")}
          disabled={isUpdatingStatus}
        >
          <Text style={styles.actionButtonText}>
            {isUpdatingStatus ? "Updating..." : "Start Ride"}
          </Text>
        </TouchableOpacity>
      );
    }

    if (assignedRide.status === "IN_PROGRESS") {
      return (
        <TouchableOpacity
          style={[
            styles.completeButton,
            isUpdatingStatus ? styles.disabledButton : null,
          ]}
          onPress={() => updateRideStatus("COMPLETED")}
          disabled={isUpdatingStatus}
        >
          <Text style={styles.actionButtonText}>
            {isUpdatingStatus ? "Updating..." : "Complete Ride"}
          </Text>
        </TouchableOpacity>
      );
    }

    return null;
  }

  return (
    <ScrollView contentContainerStyle={styles.container}>
      <Text style={styles.title}>Assigned Ride</Text>

      <Text style={styles.subtitle}>
        Testing as the configured Explore PNG driver.
      </Text>

      <View style={styles.testDriverBox}>
        <Text style={styles.testDriverLabel}>Active test driver</Text>
        <Text style={styles.testDriverName}>Test Driver Two</Text>
      </View>

      <TouchableOpacity
        style={[styles.primaryButton, isLoading ? styles.disabledButton : null]}
        onPress={() => findAssignedRide()}
        disabled={isLoading}
      >
        <Text style={styles.primaryButtonText}>
          {isLoading ? "Checking..." : "Find Assigned Ride"}
        </Text>
      </TouchableOpacity>

      <TouchableOpacity
        style={[styles.refreshButton, isLoading ? styles.disabledButton : null]}
        onPress={() => findAssignedRide()}
        disabled={isLoading}
      >
        <Text style={styles.refreshButtonText}>
          {isLoading ? "Refreshing..." : "Refresh Assigned Ride"}
        </Text>
      </TouchableOpacity>

      {message ? (
        <View
          style={assignedRide ? styles.successMessageBox : styles.messageBox}
        >
          <Text
            style={
              assignedRide ? styles.successMessageText : styles.messageText
            }
          >
            {message}
          </Text>
        </View>
      ) : null}

      {assignedRide ? (
        <View style={styles.card}>
          <Text style={styles.cardTitle}>Assigned Ride Found</Text>

          <Text style={styles.status}>{assignedRide.status}</Text>
          <Text style={styles.statusMessage}>
            {getDriverActionMessage(assignedRide.status)}
          </Text>

          <View style={styles.autoRefreshBox}>
            <Text style={styles.autoRefreshText}>
              {getAutoRefreshMessage(assignedRide.status)}
            </Text>
          </View>

          <View
            style={[
              styles.trackingBox,
              trackingError ? styles.trackingErrorBox : null,
            ]}
          >
            <Text style={styles.trackingTitle}>Live GPS Tracking</Text>

            <Text
              style={[
                styles.trackingText,
                trackingError ? styles.trackingErrorText : null,
              ]}
            >
              {Platform.OS === "web"
                ? "Live GPS tracking is disabled on web. Use the Android app for live location sharing."
                : trackingStatus === "requesting-permission"
                  ? "Requesting location permission..."
                  : trackingError
                    ? trackingError
                    : isTracking
                      ? "Live location sharing is active."
                      : "Waiting for live GPS tracking to start."}
            </Text>

            <RideMap
              driverLocation={driverMapLocation}
              pickupLocation={pickupMapLocation}
              dropoffLocation={dropoffMapLocation}
              activeDestination={activeDestination}
              route={drivingRoute}
              routeLoading={routeLoading}
              routeError={routeError}
              navigationMode={navigationMode}
            />

            {latestLocation ? (
              <>
                <Text style={styles.trackingCoordinate}>
                  Accuracy:{" "}
                  {latestLocation.accuracy === null
                    ? "Unavailable"
                    : `${latestLocation.accuracy.toFixed(1)} m`}
                </Text>

                <Text style={styles.trackingTimestamp}>
                  Last updated:{" "}
                  {new Date(latestLocation.recordedAt).toLocaleTimeString()}
                </Text>
              </>
            ) : null}
          </View>

          {renderActionButton()}

          <View style={styles.divider} />

          <Text style={styles.detailLabel}>Ride ID</Text>
          <Text style={styles.detailText}>{assignedRide.id}</Text>

          <Text style={styles.detailLabel}>Rider</Text>
          <Text style={styles.detailText}>{assignedRide.riderName}</Text>
          <Text style={styles.detailText}>{assignedRide.riderPhone}</Text>

          <Text style={styles.detailLabel}>Pickup</Text>
          <Text style={styles.detailText}>{assignedRide.pickupLabel}</Text>
          <Text style={styles.detailText}>{assignedRide.pickupAddress}</Text>

          <Text style={styles.detailLabel}>Dropoff</Text>
          <Text style={styles.detailText}>{assignedRide.dropoffLabel}</Text>
          <Text style={styles.detailText}>{assignedRide.dropoffAddress}</Text>

          <Text style={styles.detailLabel}>Estimated Fare</Text>
          <Text style={styles.detailText}>
            K{assignedRide.estimatedFare ?? "-"}
          </Text>

          {assignedRide.vehicle ? (
            <>
              <View style={styles.divider} />
              <Text style={styles.detailLabel}>Vehicle</Text>
              <Text style={styles.detailText}>
                {assignedRide.vehicle.color} {assignedRide.vehicle.make}{" "}
                {assignedRide.vehicle.model}
              </Text>
              <Text style={styles.detailText}>
                Plate: {assignedRide.vehicle.plateNumber}
              </Text>
            </>
          ) : null}
        </View>
      ) : null}
    </ScrollView>
  );
}

import { API_BASE_URL } from "@/constants/api";
import * as Location from "expo-location";
import { Platform } from "react-native";
import { useEffect, useRef, useState } from "react";

type TrackingStatus =
  "idle" | "requesting-permission" | "tracking" | "permission-denied" | "error";

type LatestLocation = {
  latitude: number;
  longitude: number;
  accuracy: number | null;
  heading: number | null;
  speed: number | null;
  recordedAt: string;
};

type LocationSample = {
  latitude: number;
  longitude: number;
  accuracy: number | null;
  heading: number | null;
  speed: number | null;
  timestamp: number;
};

type UseDriverLocationTrackingOptions = {
  driverId: string;
  rideId?: string;
  enabled: boolean;
};

export function useDriverLocationTracking({
  driverId,
  rideId,
  enabled,
}: UseDriverLocationTrackingOptions) {
  const [status, setStatus] = useState<TrackingStatus>("idle");
  const [errorMessage, setErrorMessage] = useState("");
  const [latestLocation, setLatestLocation] = useState<LatestLocation | null>(
    null,
  );

  const isPostingRef = useRef(false);

  useEffect(() => {
    let cancelled = false;
    let nativeSubscription: Location.LocationSubscription | null = null;
    let webWatchId: number | null = null;

    const normalizedDriverId = driverId.trim();
    const apiKey =
      process.env.EXPO_PUBLIC_DRIVER_LOCATION_API_KEY?.trim() ?? "";

    async function submitLocation(sample: LocationSample) {
      if (cancelled || isPostingRef.current) {
        return;
      }

      isPostingRef.current = true;

      try {
        const payload = {
          driverId: normalizedDriverId,
          rideId: rideId || undefined,
          latitude: sample.latitude,
          longitude: sample.longitude,
          accuracy: sample.accuracy ?? undefined,
          heading:
            sample.heading !== null && sample.heading >= 0
              ? sample.heading
              : undefined,
          speed:
            sample.speed !== null && sample.speed >= 0
              ? sample.speed
              : undefined,
          recordedAt: new Date(sample.timestamp).toISOString(),
        };

        const response = await fetch(
          `${API_BASE_URL}/driver-locations/upsert`,
          {
            method: "POST",
            headers: {
              "Content-Type": "application/json",
              Authorization: `Bearer ${apiKey}`,
              "X-Driver-Id": normalizedDriverId,
            },
            body: JSON.stringify(payload),
          },
        );

        const data = await response.json().catch(() => null);

        if (!response.ok) {
          if (response.status === 401) {
            throw new Error(
              "Driver location authentication failed. Restart Expo after checking the configured API key.",
            );
          }

          throw new Error(
            data?.message || "The backend rejected the location update.",
          );
        }

        if (!cancelled) {
          setLatestLocation({
            latitude: sample.latitude,
            longitude: sample.longitude,
            accuracy: sample.accuracy,
            heading: sample.heading,
            speed: sample.speed,
            recordedAt: payload.recordedAt,
          });

          setStatus("tracking");
          setErrorMessage("");
        }
      } catch (error) {
        if (!cancelled) {
          setStatus("error");
          setErrorMessage(
            error instanceof Error
              ? error.message
              : "Could not submit the driver location.",
          );
        }
      } finally {
        isPostingRef.current = false;
      }
    }

    async function startNativeTracking() {
      setStatus("requesting-permission");

      const permission = await Location.requestForegroundPermissionsAsync();

      if (cancelled) {
        return;
      }

      if (permission.status !== "granted") {
        setStatus("permission-denied");
        setErrorMessage(
          "Location permission is required to share the driver position.",
        );
        return;
      }

      const servicesEnabled = await Location.hasServicesEnabledAsync();

      if (!servicesEnabled) {
        setStatus("error");
        setErrorMessage(
          "Location services are turned off. Enable GPS and try again.",
        );
        return;
      }

      nativeSubscription = await Location.watchPositionAsync(
        {
          accuracy: Location.Accuracy.High,
          timeInterval: 10_000,
          distanceInterval: 10,
        },
        (location) => {
          void submitLocation({
            latitude: location.coords.latitude,
            longitude: location.coords.longitude,
            accuracy: location.coords.accuracy,
            heading: location.coords.heading,
            speed: location.coords.speed,
            timestamp: location.timestamp,
          });
        },
      );

      if (!cancelled) {
        setStatus("tracking");
      }
    }

    function startWebTracking() {
      if (typeof navigator === "undefined" || !navigator.geolocation) {
        setStatus("error");
        setErrorMessage("Location tracking is not supported by this browser.");
        return;
      }

      setStatus("requesting-permission");

      webWatchId = navigator.geolocation.watchPosition(
        (position) => {
          if (!cancelled) {
            setStatus("tracking");
            setErrorMessage("");
          }

          void submitLocation({
            latitude: position.coords.latitude,
            longitude: position.coords.longitude,
            accuracy: position.coords.accuracy,
            heading: position.coords.heading,
            speed: position.coords.speed,
            timestamp: position.timestamp,
          });
        },
        (error) => {
          if (cancelled) {
            return;
          }

          if (error.code === error.PERMISSION_DENIED) {
            setStatus("permission-denied");
            setErrorMessage(
              "Location permission was denied in the browser. Allow location access and refresh the page.",
            );
            return;
          }

          setStatus("error");
          setErrorMessage(
            error.message || "The browser could not retrieve your location.",
          );
        },
        {
          enableHighAccuracy: true,
          maximumAge: 5_000,
          timeout: 20_000,
        },
      );
    }

    function startTracking() {
      if (!enabled || !normalizedDriverId) {
        setStatus("idle");
        setErrorMessage("");
        setLatestLocation(null);
        return;
      }

      if (!apiKey) {
        setStatus("error");
        setErrorMessage(
          "Driver location authentication is required. Restart Expo after configuring the API key.",
        );
        return;
      }

      setErrorMessage("");

      if (Platform.OS === "web") {
        startWebTracking();
        return;
      }

      void startNativeTracking();
    }

    startTracking();

    return () => {
      cancelled = true;

      if (webWatchId !== null && typeof navigator !== "undefined") {
        navigator.geolocation.clearWatch(webWatchId);
      }

      if (Platform.OS !== "web") {
        nativeSubscription?.remove();
      }
    };
  }, [driverId, enabled, rideId]);

  return {
    status,
    errorMessage,
    latestLocation,
    isTracking: status === "tracking",
  };
}

function buildReadableAddress(
  address: Location.LocationGeocodedAddress | null,
): string {
  if (!address) {
    return '';
  }

  const values = [
    address.name,
    address.street,
    address.district,
    address.subregion,
    address.city,
    address.region,
    address.country,
  ].filter(
    (value, index, array): value is string =>
      Boolean(value) &&
      array.indexOf(value) === index,
  );

  return values.join(', ');
}

    const current =
      await Location.getForegroundPermissionsAsync();

    if (current.status === 'granted') {
      return true;
    }

    const requested =
      await Location.requestForegroundPermissionsAsync();

    if (requested.status !== 'granted') {

import {
  useCallback,
  useEffect,
  useRef,
  useState,
} from 'react';

const API_BASE_URL = (
  process.env.EXPO_PUBLIC_API_BASE_URL ??
  'https://explore-png-api.onrender.com'
).replace(/\/+$/, '');

const DEFAULT_POLL_INTERVAL_MS = 5000;

export type RideDriverLocation = {
  rideId: string;
  driverId: string;
  latitude: number;
  longitude: number;
  accuracy: number | null;
  heading: number | null;
  speed: number | null;
  recordedAt: string;
  isFresh: boolean;
};

type UseRideDriverLocationOptions = {
  rideId?: string | null;
  enabled: boolean;
  pollIntervalMs?: number;
};

type UseRideDriverLocationResult = {
  location: RideDriverLocation | null;
  isLoading: boolean;
  error: string | null;
  refresh: () => Promise<void>;
};

function asNullableNumber(value: unknown) {
  return typeof value === 'number' &&
    Number.isFinite(value)
    ? value
    : null;
}

function parseDriverLocation(
  value: unknown,
): RideDriverLocation | null {
  if (
    !value ||
    typeof value !== 'object'
  ) {
    return null;
  }

  const candidate = value as Record<string, unknown>;

  const latitude = asNullableNumber(
    candidate.latitude,
  );
  const longitude = asNullableNumber(
    candidate.longitude,
  );

  if (
    latitude == null ||
    longitude == null ||
    typeof candidate.rideId !== 'string' ||
    typeof candidate.driverId !== 'string' ||
    typeof candidate.recordedAt !== 'string'
  ) {
    return null;
  }

  return {
    rideId: candidate.rideId,
    driverId: candidate.driverId,
    latitude,
    longitude,
    accuracy: asNullableNumber(
      candidate.accuracy,
    ),
    heading: asNullableNumber(
      candidate.heading,
    ),
    speed: asNullableNumber(
      candidate.speed,
    ),
    recordedAt: candidate.recordedAt,
    isFresh: candidate.isFresh === true,
  };
}

export function useRideDriverLocation({
  rideId,
  enabled,
  pollIntervalMs = DEFAULT_POLL_INTERVAL_MS,
}: UseRideDriverLocationOptions): UseRideDriverLocationResult {
  const [location, setLocation] =
    useState<RideDriverLocation | null>(null);
  const [isLoading, setIsLoading] =
    useState(false);
  const [error, setError] =
    useState<string | null>(null);

  const abortControllerRef =
    useRef<AbortController | null>(null);
  const requestInFlightRef =
    useRef(false);

  const refresh = useCallback(async () => {
    const cleanRideId = rideId?.trim();

    if (
      !enabled ||
      !cleanRideId ||
      requestInFlightRef.current
    ) {
      return;
    }

    requestInFlightRef.current = true;

    abortControllerRef.current?.abort();

    const controller = new AbortController();
    abortControllerRef.current = controller;

    setIsLoading(true);

    try {
      const response = await fetch(
        `${API_BASE_URL}/rides/${encodeURIComponent(
          cleanRideId,
        )}/location`,
        {
          signal: controller.signal,
        },
      );

      /*
       * A 404 is expected for a short period when:
       * - a Driver has only just been assigned, or
       * - that Driver has not sent the first GPS ping yet.
       *
       * Do not turn that normal transitional state into a
       * Rider-facing error.
       */
      if (response.status === 404) {
        setError(null);
        return;
      }

      if (!response.ok) {
        throw new Error(
          `Driver location request failed (${response.status}).`,
        );
      }

      const data: unknown = await response.json();
      const parsed = parseDriverLocation(data);

      if (!parsed) {
        throw new Error(
          'Driver location response was invalid.',
        );
      }

      setLocation(parsed);
      setError(null);
    } catch (caught) {
      if (
        caught instanceof Error &&
        caught.name === 'AbortError'
      ) {
        return;
      }

      setError(
        caught instanceof Error
          ? caught.message
          : 'Could not load the Driver location.',
      );
    } finally {
      if (
        abortControllerRef.current === controller
      ) {
        abortControllerRef.current = null;
      }

      requestInFlightRef.current = false;
      setIsLoading(false);
    }
  }, [enabled, rideId]);

  useEffect(() => {
    if (!enabled || !rideId?.trim()) {
      abortControllerRef.current?.abort();
      abortControllerRef.current = null;
      requestInFlightRef.current = false;

      setLocation(null);
      setError(null);
      setIsLoading(false);

      return;
    }

    void refresh();

    const interval = setInterval(() => {
      void refresh();
    }, pollIntervalMs);

    return () => {
      clearInterval(interval);

      abortControllerRef.current?.abort();
      abortControllerRef.current = null;
      requestInFlightRef.current = false;
    };
  }, [
    enabled,
    pollIntervalMs,
    refresh,
    rideId,
  ]);

  return {
    location,
    isLoading,
    error,
    refresh,
  };
}

import {
  BadRequestException,
  ConflictException,
  Injectable,
  UnauthorizedException,
} from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import { Prisma } from '@prisma/client';
import * as bcrypt from 'bcrypt';
import { randomBytes } from 'crypto';
import { PrismaService } from '../prisma/prisma.service';
import { RiderLoginDto } from './dto/rider-login.dto';
import { RiderRegisterDto } from './dto/rider-register.dto';

const ACCESS_TOKEN_LIFETIME_SECONDS = 12 * 60 * 60;
const PASSWORD_HASH_ROUNDS = 12;

@Injectable()
export class RiderAuthService {
  private readonly dummyPasswordHash = bcrypt.hash(
    randomBytes(32).toString('hex'),
    PASSWORD_HASH_ROUNDS,
  );

  constructor(
    private readonly prisma: PrismaService,
    private readonly jwtService: JwtService,
  ) {}

  async register(dto: RiderRegisterDto) {
    const fullName = dto.fullName.trim();
    const phone = dto.phone.trim();
    const email = dto.email?.trim().toLowerCase();

    if (!fullName) {
      throw new BadRequestException('Rider full name is required');
    }

    const existingAccount = await this.prisma.riderAccount.findFirst({
      where: {
        OR: [{ phone }, ...(email ? [{ email }] : [])],
      },
      select: {
        phone: true,
        email: true,
      },
    });

    if (existingAccount) {
      throw new ConflictException(
        'A Rider account already exists for that phone or email',
      );
    }

    const passwordHash = await bcrypt.hash(dto.password, PASSWORD_HASH_ROUNDS);

    try {
      const rider = await this.prisma.riderAccount.create({
        data: {
          fullName,
          phone,
          email,
          passwordHash,
        },
      });

      return this.issueAccessToken(rider);
    } catch (error: unknown) {
      if (
        error instanceof Prisma.PrismaClientKnownRequestError &&
        error.code === 'P2002'
      ) {
        throw new ConflictException(
          'A Rider account already exists for that phone or email',
        );
      }

      throw error;
    }
  }

  async login(dto: RiderLoginDto) {
    const phone = dto.phone.trim();

    const rider = await this.prisma.riderAccount.findUnique({
      where: {
        phone,
      },
    });

    const passwordHash = rider?.passwordHash ?? (await this.dummyPasswordHash);

    const passwordMatches = await bcrypt.compare(dto.password, passwordHash);

    if (!rider || !rider.isActive || !passwordMatches) {
      throw new UnauthorizedException('Invalid Rider credentials');
    }

    const updatedRider = await this.prisma.riderAccount.update({
      where: {
        id: rider.id,
      },
      data: {
        lastLoginAt: new Date(),
      },
    });

    return this.issueAccessToken(updatedRider);
  }

  private async issueAccessToken(rider: {
    id: string;
    fullName: string;
    phone: string;
    email: string | null;
    tokenVersion: number;
  }) {
    const accessToken = await this.jwtService.signAsync({
      sub: rider.id,
      phone: rider.phone,
      tokenVersion: rider.tokenVersion,
      tokenType: 'rider-access',
    });

    return {
      accessToken,
      tokenType: 'Bearer',
      expiresInSeconds: ACCESS_TOKEN_LIFETIME_SECONDS,
      rider: {
        id: rider.id,
        fullName: rider.fullName,
        phone: rider.phone,
        email: rider.email,
      },
    };
  }
}

      where: {

    });

    const passwordHash = driver?.passwordHash ?? (await this.dummyPasswordHash);

    const passwordMatches = await bcrypt.compare(dto.password, passwordHash);

      !driver.isAuthActive ||

        id: true,

        id: true,
        fullName: true,
        phone: true,
        email: true,
        status: true,
        isAuthActive: true,
        tokenVersion: true,

import { Injectable, UnauthorizedException } from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import * as bcrypt from 'bcrypt';
import { randomBytes } from 'crypto';
import { PrismaService } from '../prisma/prisma.service';
import { AdminLoginDto } from './dto/admin-login.dto';

const ACCESS_TOKEN_LIFETIME_SECONDS = 15 * 60;

@Injectable()
export class AdminAuthService {
  private readonly dummyPasswordHash = bcrypt.hash(
    randomBytes(32).toString('hex'),
    12,
  );

  constructor(
    private readonly prisma: PrismaService,
    private readonly jwtService: JwtService,
  ) {}

  async login(dto: AdminLoginDto) {
    const email = dto.email.trim().toLowerCase();

    const admin = await this.prisma.adminAccount.findUnique({
      where: { email },
    });

    const passwordHash = admin?.passwordHash ?? (await this.dummyPasswordHash);

    const passwordMatches = await bcrypt.compare(dto.password, passwordHash);

    if (!admin || !admin.isActive || !passwordMatches) {
      throw new UnauthorizedException('Invalid administrator credentials');
    }

    await this.prisma.adminAccount.update({
      where: { id: admin.id },
      data: { lastLoginAt: new Date() },
    });

    const accessToken = await this.jwtService.signAsync({
      sub: admin.id,
      email: admin.email,
      role: admin.role,
      tokenVersion: admin.tokenVersion,
      tokenType: 'admin-access',
    });

    return {
      accessToken,
      tokenType: 'Bearer',
      expiresInSeconds: ACCESS_TOKEN_LIFETIME_SECONDS,
      admin: {
        id: admin.id,
        email: admin.email,
        fullName: admin.fullName,
        role: admin.role,
      },
    };
  }
}

import {
  CanActivate,
  ExecutionContext,
  Injectable,
  UnauthorizedException,
} from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import { AdminRole } from '@prisma/client';
import { Request } from 'express';
import { PrismaService } from '../../../prisma/prisma.service';

type AdminAccessTokenPayload = {
  sub: string;
  email: string;
  role: AdminRole;
  tokenVersion: number;
  tokenType: 'admin-access';
};

export type AuthenticatedAdmin = {
  id: string;
  email: string;
  fullName: string;
  role: AdminRole;
};

export type AdminAuthenticatedRequest = Request & {
  admin?: AuthenticatedAdmin;
};

@Injectable()
export class AdminJwtAuthGuard implements CanActivate {
  constructor(
    private readonly jwtService: JwtService,
    private readonly prisma: PrismaService,
  ) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context
      .switchToHttp()
      .getRequest<AdminAuthenticatedRequest>();

    const authorization = request.header('authorization');

    if (!authorization?.startsWith('Bearer ')) {
      throw new UnauthorizedException(
        'Administrator authentication is required',
      );
    }

    const token = authorization.slice('Bearer '.length).trim();

    if (!token) {
      throw new UnauthorizedException('Administrator authentication failed');
    }

    try {
      const payload =
        await this.jwtService.verifyAsync<AdminAccessTokenPayload>(token);

      if (
        payload.tokenType !== 'admin-access' ||
        !payload.sub ||
        !Number.isInteger(payload.tokenVersion)
      ) {
        throw new UnauthorizedException();
      }

      const admin = await this.prisma.adminAccount.findUnique({
        where: { id: payload.sub },
        select: {
          id: true,
          email: true,
          fullName: true,
          role: true,
          isActive: true,
          tokenVersion: true,
        },
      });

      if (
        !admin ||
        !admin.isActive ||
        admin.tokenVersion !== payload.tokenVersion
      ) {
        throw new UnauthorizedException();
      }

      request.admin = {
        id: admin.id,
        email: admin.email,
        fullName: admin.fullName,
        role: admin.role,
      };

      return true;
    } catch {
      throw new UnauthorizedException('Administrator authentication failed');
    }
  }
}

      throw new BadRequestException(
        `Driver can only be assigned while ride is REQUESTED. Current status is ${ride.status}`,

    if (ride.driverId || ride.vehicleId) {
      throw new BadRequestException(

    const driver = await this.prisma.driver.findUnique({
      where: {
        id: dto.driverId,
      },
      include: {
        vehicles: {
          where: {
            status: 'ACTIVE',
          },
          take: 1,

    if (!driver) {
      throw new NotFoundException('Driver not found');
    }

    if (driver.status !== 'APPROVED') {
      throw new BadRequestException(
        `Driver is not approved. Current status is ${driver.status}`,
      );
    }

    if (!driver.isOnline) {
      throw new BadRequestException('Driver is not online');

    if (!driver.isAvailable) {
      throw new BadRequestException('Driver is not available');
    }

      throw new BadRequestException(
        'Driver is not assigned to the ride service zone',

    if (!activeVehicle) {
      throw new BadRequestException('Selected driver has no active vehicle');
    }

      /*
       * Atomically claim the driver.
       *
       * If another assignment claims the same driver first,
       * this update affects zero records.
       */
      const claimedDriver = await tx.driver.updateMany({
        where: {
          id: driver.id,
          status: 'APPROVED',
          isOnline: true,
          isAvailable: true,

        data: {
          isAvailable: false,

        throw new BadRequestException(
          'Driver is no longer available and may have been assigned to another ride',
        );
      }

      /*
       * Atomically claim the ride.
       *
       * If another dispatcher assigns this ride first,
       * this update affects zero records.
       */
      const claimedRide = await tx.ride.updateMany({
        where: {
          id,
          status: RideStatus.REQUESTED,
          driverId: null,
          vehicleId: null,
        },
        data: {
          driverId: driver.id,
          vehicleId: activeVehicle.id,
          status: RideStatus.ACCEPTED,
        },
      });

      if (claimedRide.count !== 1) {
        throw new BadRequestException(
          'Ride is no longer available for driver assignment',
        );
      }

      const assignedRide = await tx.ride.findUnique({
        where: {
          id,
        },
        include: this.defaultInclude(),
      });

      if (!assignedRide) {
        throw new NotFoundException('Ride could not be found after assignment');

      return assignedRide;

  async updateStatus(id: string, dto: UpdateRideStatusDto) {
    const ride = await this.findOne(id);

    assertRideStatusTransition(ride.status, dto.status);

    const releasingStatuses: RideStatus[] = [
      RideStatus.COMPLETED,
      RideStatus.CANCELLED,
      RideStatus.NO_DRIVER_AVAILABLE,
    ];

    const driverIdToRelease =
      ride.driverId && releasingStatuses.includes(dto.status)
        ? ride.driverId
        : undefined;

    const shouldCreateSettlement = dto.status === RideStatus.COMPLETED;

    const finalFareAmount = shouldCreateSettlement
      ? (ride.finalFare ?? ride.estimatedFare)
      : undefined;

    if (
      shouldCreateSettlement &&
      (finalFareAmount === undefined ||
        finalFareAmount === null ||
        finalFareAmount <= 0)
    ) {
      throw new BadRequestException(
        'Cannot complete ride without a valid finalFare or estimatedFare',
      );
    }

    return this.prisma.$transaction(async (tx) => {
      await tx.ride.update({
        where: {
          id,
        },
        data: {
          status: dto.status,
          notes: dto.notes ?? undefined,
          ...(shouldCreateSettlement ? { finalFare: finalFareAmount } : {}),
        },
      });

      if (shouldCreateSettlement && finalFareAmount) {
        const finalFareDecimal = new Prisma.Decimal(finalFareAmount);

        const driverShare = finalFareDecimal.times(0.8).toDecimalPlaces(2);

        const companyShare = finalFareDecimal.times(0.2).toDecimalPlaces(2);

        await tx.rideSettlement.upsert({
          where: {
            rideId: id,
          },
          update: {
            finalFare: finalFareDecimal,
            driverShare,
            companyShare,
            driverId: ride.driverId,
            vehicleId: ride.vehicleId,
            status: RideSettlementStatus.PENDING,
          },
          create: {
            rideId: id,
            receiptNumber: this.generateReceiptNumber(),
            driverId: ride.driverId,
            vehicleId: ride.vehicleId,
            finalFare: finalFareDecimal,
            driverShare,
            companyShare,
            status: RideSettlementStatus.PENDING,
          },
        });
      }

      if (driverIdToRelease) {
        await tx.driver.updateMany({
          where: {
            id: driverIdToRelease,
            status: 'APPROVED',
            isOnline: true,
          },
          data: {
            isAvailable: true,
          },
        });
      }

      return tx.ride.findUnique({
        where: {
          id,
        },
        include: this.defaultInclude(),
      });
    });
  }
